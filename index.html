CREATE OR REPLACE FUNCTION public.get_section_admins_for_messaging(p_section_id BIGINT) RETURNS TABLE(user_id UUID, full_name TEXT, email TEXT, admin_source TEXT) LANGUAGE plpgsql STABLE SECURITY DEFINER SET search_path=public,pg_temp AS $$ DECLARE v_course_id BIGINT := p_section_id; BEGIN IF auth.uid() IS NULL OR p_section_id IS NULL THEN RAISE EXCEPTION 'DENY ACCESS'; END IF; -- Resolve any chapter/material/course node to its nearest course ancestor. WITH RECURSIVE ancestors AS ( SELECT s.id, s.parent_id, s.type, 0 AS depth FROM site_sections s WHERE s.id=p_section_id UNION ALL SELECT p.id, p.parent_id, p.type, a.depth+1 FROM site_sections p JOIN ancestors a ON p.id=a.parent_id WHERE a.depth < 20 ) SELECT id INTO v_course_id FROM ancestors WHERE LOWER(COALESCE(type,''))='course' ORDER BY depth LIMIT 1; -- Course admin first. Accept both admin and owner assignments. RETURN QUERY SELECT sa.user_id, COALESCE(p.full_name,p.email,'أدمن الكورس')::TEXT, p.email::TEXT, 'course'::TEXT FROM section_admins sa JOIN profiles p ON p.id=sa.user_id WHERE sa.section_id=v_course_id AND sa.is_active=true AND sa.role IN ('admin','owner') ORDER BY COALESCE(p.full_name,p.email,''); IF FOUND THEN RETURN; END IF; -- No course admin: platform admin fallback. RETURN QUERY SELECT p.id, COALESCE(p.full_name,p.email,'أدمن المنصة')::TEXT, p.email::TEXT, 'platform'::TEXT FROM profiles p WHERE p.role='admin' ORDER BY COALESCE(p.full_name,p.email,''); END; $$; GRANT EXECUTE ON FUNCTION public.get_section_admins_for_messaging(BIGINT) TO authenticated; DROP FUNCTION IF EXISTS public.get_or_create_direct_room(UUID,BIGINT); CREATE OR REPLACE FUNCTION public.get_or_create_direct_room(p_other_user_id UUID,p_section_id BIGINT) RETURNS chat_rooms LANGUAGE plpgsql SECURITY DEFINER SET search_path=public,pg_temp AS $$ DECLARE v_me UUID := auth.uid(); v_room chat_rooms; v_course_id BIGINT := p_section_id; v_is_course_admin BOOLEAN := false; v_is_platform_admin BOOLEAN := false; BEGIN IF v_me IS NULL OR p_other_user_id IS NULL OR p_other_user_id=v_me OR p_section_id IS NULL THEN RAISE EXCEPTION 'DENY ACCESS'; END IF; WITH RECURSIVE ancestors AS ( SELECT s.id, s.parent_id, s.type, 0 AS depth FROM site_sections s WHERE s.id=p_section_id UNION ALL SELECT p.id, p.parent_id, p.type, a.depth+1 FROM site_sections p JOIN ancestors a ON p.id=a.parent_id WHERE a.depth < 20 ) SELECT id INTO v_course_id FROM ancestors WHERE LOWER(COALESCE(type,''))='course' ORDER BY depth LIMIT 1; SELECT EXISTS(SELECT 1 FROM section_admins sa WHERE sa.user_id=p_other_user_id AND sa.section_id=v_course_id AND sa.is_active=true AND sa.role IN ('admin','owner')) INTO v_is_course_admin; SELECT EXISTS(SELECT 1 FROM profiles p WHERE p.id=p_other_user_id AND p.role='admin') INTO v_is_platform_admin; IF NOT v_is_course_admin AND NOT v_is_platform_admin THEN RAISE EXCEPTION 'DENY ACCESS: الطرف المطلوب ليس أدمن الكورس أو أدمن المنصة'; END IF; PERFORM pg_advisory_xact_lock(hashtextextended(LEAST(v_me::TEXT,p_other_user_id::TEXT)||':'||GREATEST(v_me::TEXT,p_other_user_id::TEXT)||':'||v_course_id::TEXT,0)); SELECT cr.* INTO v_room FROM chat_rooms cr WHERE cr.section_id=v_course_id AND cr.room_type='direct' AND EXISTS(SELECT 1 FROM chat_room_participants x WHERE x.room_id=cr.id AND x.user_id=v_me) AND EXISTS(SELECT 1 FROM chat_room_participants y WHERE y.room_id=cr.id AND y.user_id=p_other_user_id) ORDER BY cr.id LIMIT 1; IF FOUND THEN RETURN v_room; END IF; INSERT INTO chat_rooms(room_name,section_id,room_type) VALUES('محادثة خاصة - '||COALESCE((SELECT title FROM site_sections WHERE id=v_course_id),'الكورس'),'direct') RETURNING * INTO v_room; INSERT INTO chat_room_participants(room_id,user_id) VALUES(v_room.id,v_me),(v_room.id,p_other_user_id); RETURN v_room; END; $$; GRANT EXECUTE ON FUNCTION public.get_or_create_direct_room(UUID,BIGINT) TO authenticated;
